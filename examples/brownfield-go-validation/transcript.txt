+func TestValidateTitle(t *testing.T) {                                         
+   cases := []struct {                                                         
+       in, want string                                                         
+       ok       bool                                                           
+   }{                                                                          
+       {"buy milk", "buy milk", true},                                         
+       {"  padded  ", "padded", true},                                         
+       {"", "", false},                                                        
+       {"   ", "", false},                                                     
+       {strings.Repeat("a", MaxTitleLen), strings.Repeat("a", MaxTitleLen),    
true},                                                                          
+       {strings.Repeat("a", MaxTitleLen+1), "", false},                        
+       {strings.Repeat("é", MaxTitleLen), strings.Repeat("é", MaxTitleLen),    
true}, // counts characters, not bytes                                          
+   }                                                                           
+   for _, c := range cases {                                                   
+       got, err := ValidateTitle(c.in)                                         
+       if (err == nil) != c.ok || got != c.want {                              
+           t.Errorf("ValidateTitle(%q) = %q, %v; want %q, ok=%v", c.in, got,   
err, c.want, c.ok)                                                              
+       }                                                                       
+   }                                                                           
+}
