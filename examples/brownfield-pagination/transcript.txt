╭─────────────────────────────────── Impact ───────────────────────────────────╮
│ GET /notes in notes/app.py loads every note, then filters by tag in Python.  │
│ Paging on top of that would be wrong: a page of 50 could come back with      │

│ LIMIT is applied. No other endpoint or the schema needs to change.           │

+def test_default_page_size_and_cursor(client):                                 
+    _make(client, DEFAULT_PAGE_SIZE + 5)                                       
+    first = client.get("/notes")                                               
+    assert len(first.json()) == DEFAULT_PAGE_SIZE                              
+    cursor = first.headers["X-Next-Cursor"]                                    
+    second = client.get("/notes", params={"after": cursor})                    
+    assert len(second.json()) == 5                                             
+    assert "X-Next-Cursor" not in second.headers                               
+                                                                               
+                                                                               
+def test_walks_every_note_exactly_once(client):                                
+    ids = _make(client, 23)                                                    
+    seen, cursor = [], None                                                    
+    while True:                                                                
+        r = client.get("/notes", params={"limit": 5, **({"after": cursor} if   
cursor else {})})                                                               
+        seen += [n["id"] for n in r.json()]                                    
+        cursor = r.headers.get("X-Next-Cursor")                                
+        if not cursor:                                                         
+            break                                                              
+    assert seen == ids                                                         
+                                                                               
+                                                                               
+def test_tag_filter_is_applied_before_paging(client):                          
+    _make(client, 10, tags=["other"])

+    rest = client.get("/notes", params={"tag": "x", "limit": 2, "after":       
r.headers["X-Next-Cursor"]})                                                    
+    assert [n["id"] for n in rest.json()] == wanted[2:]
